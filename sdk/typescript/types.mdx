## SessionBlock

One block in a [debugger](/debugger/introduction) session, as returned by `client.rolloutSessions.listBlocks(...)`.

```typescript
type SessionBlockType = "trace" | "evaluation" | "text" | "command";

interface TraceBlockContent {
  traceId: string;
  note?: string | null;
}

interface EvaluationBlockContent {
  evaluationId: string;
  note?: string | null;
}

interface TextBlockContent {
  text: string;
}

interface CommandBlockContent {
  command: string;
  args: string[];
  exitCode: number;
  output?: string | null;
  stderr?: string | null;
  reasoning?: string | null;
}

type SessionBlockContent =
  | TraceBlockContent
  | EvaluationBlockContent
  | TextBlockContent
  | CommandBlockContent;

interface SessionBlock {
  id: string;
  createdAt: string; // ISO-8601
  type: string; // one of SessionBlockType for known blocks
  content: Record<string, unknown>; // narrow with the *BlockContent interfaces
}
```
